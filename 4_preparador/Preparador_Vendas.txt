; ============================================================
; PREPARADOR VENDAS
; ============================================================
;
; Módulo de mantenimiento de vendas usadas.
;
; Función:
;   - Detectar vendas sucias en mochila
;   - Aplicar limpieza usando contenedor/target fijo
;
; No realiza movimiento.
; No selecciona objetivos de farmeo.
; Solo mantenimiento de inventario.
;
; TODO usar variables de configuracion para targetID y tileFound
; ============================================================

Preparador:
    gosub LimpiarVendas
    wait 120s
exit


sub LimpiarVendas

    set %vendasUsadas AMF
    set %targetID SBKEDPD
    set %tileFoundX 2353
    set %tileFoundY 1106
    set %tileFoundZ 7

    finditem %vendasUsadas C_ , #BACKPACKID

    if #FINDCNT > 0
    {
        set %vendasId #FINDID

        set #lTargetID %targetID
        set #lTargetKind 1

        set #LOBJECTID %vendasId
        event macro 17 0

        target 5s
        event Macro 22
    }

return
